cmake_minimum_required(VERSION 3.21)
project(VKI VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.7 REQUIRED COMPONENTS Core Gui Widgets OpenGLWidgets)
add_library(vki_core src/core/theme.cpp src/core/theme_source.cpp src/core/storage.cpp
    src/device/definition.cpp src/device/via.cpp src/device/keycodes.cpp)
target_include_directories(vki_core PUBLIC src)
target_link_libraries(vki_core PUBLIC Qt6::Core Qt6::Gui)
add_library(vki_ui src/ui/modal.cpp src/ui/modal_backdrop.cpp src/ui/modal_motion.cpp src/ui/themed_combo.cpp
    src/ui/theme_watcher.cpp src/ui/icons.cpp src/ui/keyboard_view.cpp src/ui/lighting_page.cpp
    src/ui/color_picker.cpp
    src/ui/keymap_page.cpp src/ui/window.cpp)
target_link_libraries(vki_ui PUBLIC vki_core Qt6::Widgets Qt6::OpenGLWidgets)
add_executable(vki src/main.cpp src/cli.cpp data/resources.qrc)
target_link_libraries(vki PRIVATE vki_ui)
foreach(target vki_core vki_ui vki)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()
include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    add_executable(core_tests tests/core_tests.cpp data/resources.qrc)
    target_link_libraries(core_tests PRIVATE vki_ui Qt6::Test)
    add_test(NAME core COMMAND core_tests)
    set_tests_properties(core PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    add_test(NAME cli COMMAND bash ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli_test.sh $<TARGET_FILE:vki>)
endif()
include(GNUInstallDirs)
install(TARGETS vki RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES assets/io.github.vki.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES assets/io.github.vki.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
install(FILES packaging/60-vki.rules DESTINATION ${CMAKE_INSTALL_LIBDIR}/udev/rules.d)
